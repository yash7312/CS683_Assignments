% \gbarplot{table}{y columns}{axis options}: grouped bars, one group per table row (label column "g")
\newcommand{\gbarplot}[3]{%
  \pgfplotstablegetrowsof{#1}\pgfmathtruncatemacro\gbn{\pgfplotsretval-1}%
  \begin{tikzpicture}
  \begin{axis}[gbar, xtick={0,...,\gbn}, xticklabels from table={#1}{g}, #3]
    \edef\gbcols{#2}%
    \expandafter\pgfplotsinvokeforeach\expandafter{\gbcols}{\addplot table[x expr=\coordindex, y=##1]{#1};}
  \end{axis}
  \end{tikzpicture}}

% \lineplot{table}{x column}{y columns}{axis options}
\newcommand{\lineplot}[4]{%
  \begin{tikzpicture}
  \begin{axis}[base, cycle list name=traces, #4]
    \edef\lpcols{#3}%
    \expandafter\pgfplotsinvokeforeach\expandafter{\lpcols}{\addplot table[x=#2, y=##1]{#1};}
  \end{axis}
  \end{tikzpicture}}
